
class UPI {
    string upiId
}

class Card {
    string cardNumber
}

class NetBanking {
    string bankName
}

class Transaction<T> {
    T paymentMethod
    double amount

    constructor(T paymentMethod, double amount) {
        this.paymentMethod = paymentMethod
        this.amount = amount
    }

    method getAmount() -> double {
        return amount
    }

    method getPaymentMethod() -> T {
        return paymentMethod
    }
}

class PaymentProcessor {
    list<Transaction<?>> transactions  

    constructor() {
        transactions = new list<>()
    }

    method addTransaction(Transaction<?> t) {
        transactions.add(t)
    }

    method processAllTransactions() {
        // Using Streams
        transactions.stream()
            .forEach(transaction -> print("Processed: " + transaction.getPaymentMethod() 
                                           + " | Amount: " + transaction.getAmount()))
    }

    method totalAmount() -> double {
        return transactions.stream()
            .mapToDouble(t -> t.getAmount())
            .sum()
    }
}


main() {
    processor = new PaymentProcessor()

    t1 = new Transaction<UPI>(new UPI("user@upi"), 500.0)
    t2 = new Transaction<Card>(new Card("1234-5678-9876-5432"), 1200.0)
    t3 = new Transaction<NetBanking>(new NetBanking("HDFC"), 2500.0)

    processor.addTransaction(t1)
    processor.addTransaction(t2)
    processor.addTransaction(t3)

    processor.processAllTransactions()
    print("Total Amount Processed: " + processor.totalAmount())
}
