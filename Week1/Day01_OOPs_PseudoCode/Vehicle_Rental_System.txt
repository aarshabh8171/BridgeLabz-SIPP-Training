INTERFACE Rentable
    METHOD rent(days)
    METHOD returnVehicle()

ABSTRACT CLASS Vehicle IMPLEMENTS Rentable
    VARIABLE vehicleId
    VARIABLE brand
    VARIABLE rentPerDay

    METHOD rent(days)
        cost = days * rentPerDay
        PRINT "Total cost = " + cost

    METHOD returnVehicle()
        PRINT "Vehicle returned."

CLASS Car EXTENDS Vehicle
    // additional attributes like seatCount

CLASS Bike EXTENDS Vehicle
    // additional attributes like type

CLASS VehicleFactory
    METHOD createVehicle(type)
        IF type == "car" THEN RETURN new Car()
        IF type == "bike" THEN RETURN new Bike()

MAIN
    factory = VehicleFactory()
    v1 = factory.createVehicle("car")
