CLASS Account
    VARIABLE accountNumber
    VARIABLE balance

    METHOD deposit(amount)
        balance = balance + amount

    METHOD withdraw(amount)
        // base: to be overridden by child classes

CLASS SavingAccount EXTENDS Account
    CONSTANT MIN_BALANCE = 1000

    METHOD withdraw(amount)
        IF balance - amount >= MIN_BALANCE THEN
            balance = balance - amount
        ELSE
            PRINT "Withdrawal denied! Minimum balance required."

CLASS CurrentAccount EXTENDS Account
    CONSTANT OVERDRAFT_LIMIT = 5000

    METHOD withdraw(amount)
        IF balance - amount >= -OVERDRAFT_LIMIT THEN
            balance = balance - amount
        ELSE
            PRINT "Withdrawal denied! Overdraft limit exceeded."

MAIN
    acc1 = SavingAccount(10001, 5000)
    acc2 = CurrentAccount(10002, 2000)

    acc1.withdraw(4500)  // Should deny
    acc2.withdraw(6000)  // Allowed due to overdraft
